<script>

    let dnd, dsr, qtn_final, current_qtn = "";

    let step = "QUOTIENT"; // QUOTIENT, MULTIPLY, SUBTRACT, DOWN

    let pointer = 0; // Quants números del dividend hem agafat


    function newGame(lv) {

        dsr = lv === 1 ? Math.floor(Math.random()*8)+2 : Math.floor(Math.random()*89)+11;

        dnd = Math.floor(Math.random()*800)+100;

        qtn_final = Math.floor(dnd / dsr).toString();

        current_qtn = "";

        

        document.getElementById('d-val').innerText = dnd;

        document.getElementById('divisor-val').innerText = dsr;

        document.getElementById('quotient-area').innerText = "";

        document.getElementById('restes-area').innerHTML = "";

        document.getElementById('input-zone').style.display = "block";

        

        step = "QUOTIENT";

        updateMsg(`Quants cops cap ${dsr} en la primera part del número?`);

    }


    function updateMsg(m) { document.getElementById('msg').innerText = m; }


    function validateStep() {

        const val = document.getElementById('user-val').value;

        const targetDigit = qtn_final[current_qtn.length];


        if (val === targetDigit) {

            current_qtn += val;

            document.getElementById('quotient-area').innerText = current_qtn;

            

            if (current_qtn === qtn_final) {

                const residu = dnd % dsr;

                addRestRow(residu);

                updateMsg("¡Molt bé! Divisió completada. El residu és " + residu);

                document.getElementById('input-zone').style.display = "none";

            } else {

                updateMsg("Correcte! Ara continua amb la següent xifra.");

            }

        } else {

            updateMsg("❌ Revisa el càlcul, aquest número no és correcte.");

        }

        document.getElementById('user-val').value = "";

    }


    function addRestRow(val) {

        const div = document.createElement('div');

        div.className = "history-row";

        div.innerText = val;

        document.getElementById('restes-area').appendChild(div);

    }

</script>
