(Dins la funció reproduir, he modificat la lògica de lectura) ... function reproduir() { window.speechSynthesis.cancel(); const btnPlay = document.getElementById('btn-play'); const speed = document.getElementById('speed').value; // 1. Preparem el text per ser llegit com un mestre // Substituïm els signes per paraules perquè la IA les digui en veu alta let textMestre = dText .replace(/, /g, " , coma , ") .replace(/\. /g, " . punt i seguit . ") .replace(/\.$/g, " . punt i final."); // Dividim el text en fragments basant-nos en els punts i comes per fer les pauses const fragments = textMestre.split(/([.,])/).filter(f => f.trim().length > 0); let i = 0; btnPlay.disabled = true; btnPlay.innerText = "Dictant..."; function llegirFragment() { if(i < fragments.length) { let fragmentText = fragments[i].trim(); // Si el fragment és només un signe, li donem una entonació especial let esSigne = fragmentText === "coma" || fragmentText === "punt i seguit" || fragmentText === "punt i final"; document.getElementById('subtitles').innerText = esSigne ? `[${fragmentText.toUpperCase()}]` : fragmentText; const ut = new SpeechSynthesisUtterance(fragmentText); ut.lang = 'ca-ES'; ut.rate = esSigne ? speed + 0.1 : speed; // Els signes els diu una mica més ràpid ut.pitch = 1; ut.onend = () => { i++; // Pausa més llarga (4 segons) després d'una frase, // i més curta (1.5 segons) després d'una coma. let tempsPausa = fragmentText === "coma" ? 1500 : 4000; setTimeout(llegirFragment, tempsPausa); }; window.speechSynthesis.speak(ut); } else { document.getElementById('subtitles').innerText = "Dictat finalitzat ✅"; btnPlay.disabled = false; btnPlay.innerText = "▶️ Tornar a dictar"; } } llegirFragment(); }
